// 打开数据库 const rrequest = indexedDB.open("SDB", 8); // 注意升级版本号 rrequest.onupgradeneeded = (e) => { e.target.result.createObjectStore("sttore", { autoIncrement: true }); }; rrequest.onsuccess = () => { const db = rrequest.result; // 创建文件选择器 const input = document.createElement("input"); input.type = "file"; input.accept = "image/*"; input.onchange = () => { const file = input.files[0]; const reader = new FileReader(); reader.onload = () => { const buffer = reader.result; // 写入 IndexedDB const tx = db.transaction("sttore", "readwrite"); const store = tx.objectStore("sttore"); const putRequest = store.put(buffer); putRequest.onsuccess = (e) => { const key = e.target.result; // 读取回来用于展示 const readTx = db.transaction("sttore", "readonly"); const readStore = readTx.objectStore("sttore"); const getRequest = readStore.get(key); getRequest.onsuccess = () => { const data = getRequest.result; const blob = new Blob([data], { type: file.type }); const url = URL.createObjectURL(blob); // 显示图片 const img = document.createElement("img"); img.src = url; img.style.maxWidth = "200px"; document.body.appendChild(img); console.log("图片已显示，Blob 长度:", data.byteLength); }; }; }; reader.readAsArrayBuffer(file); }; input.click(); };
